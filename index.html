'https://your-site/api/video-proxy (recommended)' : 'Optional key or leave blank for demo'; input.value = keys[providerId] || ''; const hint = document.createElement('div'); hint.style.fontSize='12px'; hint.style.color='var(--mut-light)'; hint.style.marginTop='6px'; hint.style.lineHeight='1.5'; if(providerId==='google' || providerId==='gemini'){ hint.textContent = 'Google Gemini setup: configure GEMINI_KEY & GEMINI_MODEL on your server or Vercel, then save your server endpoint URL here (e.g. https://your-site/api/video-proxy). Do NOT store secrets in the browser for production.'; } else { hint.textContent = provider.requiresServer ? 'Note: Most providers require server-side proxy endpoints. Store secrets server-side and call via endpoint URL here.' : 'Runs locally for demo.'; } container.appendChild(label); container.appendChild(input); container.appendChild(hint); } async function sendToVideoModel(providerId, payload){ const keys = JSON.parse(localStorage.getItem('llm_video_keys')||'{}'); const key = keys[providerId] || ''; if(providerId === 'demo'){ await new Promise(r=>setTimeout(r,2500)); return { status:'done', url:'https://cdn.example.com/demo-video.mp4' }; } if(!key || !(key.startsWith('http://') || key.startsWith('https://'))){ throw new Error('Provider requires a server endpoint URL in the provider key field.
